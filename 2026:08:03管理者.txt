  <script>
    const questionInput = document.getElementById('question-input');
    const btnShowQuestion = document.getElementById('btn-show-question');
    const btnSetWaiting = document.getElementById('btn-set-waiting');
    
    const currentStatusBadge = document.getElementById('current-status-badge');
    const screenWaiting = document.getElementById('screen-waiting');
    const screenQuestion = document.getElementById('screen-question');
    const screenQuestionText = document.getElementById('screen-question-text');
    const danmakuContainer = document.getElementById('danmaku-container');

    // スクリーン表示の切り替え
    function updateScreen(state) {
      if (!state) return;
      
      if (state.status === 'question') {
        currentStatusBadge.textContent = '質問表示中';
        currentStatusBadge.className = 'px-3 py-1 text-xs font-bold rounded-full bg-emerald-500/20 text-emerald-300 border border-emerald-500/30';
        
        screenWaiting.classList.add('hidden');
        screenQuestion.classList.remove('hidden');
        screenQuestionText.textContent = state.question_text;
        questionInput.value = state.question_text || '';
      } else {
        currentStatusBadge.textContent = '待機中';
        currentStatusBadge.className = 'px-3 py-1 text-xs font-bold rounded-full bg-yellow-500/20 text-yellow-300 border border-yellow-500/30';
        
        screenQuestion.classList.add('hidden');
        screenWaiting.classList.remove('hidden');
      }
    }

    // データベース状態の更新
    async function updateAppState(status, questionText = '') {
      const { error } = await supabase
        .from('app_state')
        .update({ status: status, question_text: questionText })
        .eq('id', 1);

      if (error) {
        alert('更新エラー: ' + error.message);
      }
    }

    // イベントリスナー
    btnShowQuestion.addEventListener('click', () => {
      const text = questionInput.value.trim();
      if (!text) {
        alert('質問テキストを入力してください');
        return;
      }
      updateAppState('question', text);
    });

    btnSetWaiting.addEventListener('click', () => {
      updateAppState('waiting', '');
    });

    // 弾幕（ニコニコ動画風テキスト流し）描画処理
    function spawnDanmaku(text) {
      const el = document.createElement('div');
      el.className = 'danmaku-item text-3xl sm:text-5xl text-white';
      el.textContent = text;

      // 画面の高さ（15% ～ 80%）にランダム配置して被りを軽減
      const topPercent = Math.floor(Math.random() * 65) + 15;
      el.style.top = `${topPercent}%`;

      danmakuContainer.appendChild(el);

      // 流れ終わったらDOMを削除
      el.addEventListener('animationend', () => {
        el.remove();
      });
    }

    // 初期化処理
    async function init() {
      const { data } = await supabase.from('app_state').select('*').eq('id', 1).single();
      if (data) updateScreen(data);
    }

    // Realtime 監視 1: app_state (画面切り替え)
    supabase
      .channel('admin:app_state')
      .on('postgres_changes', { event: '*', schema: 'public', table: 'app_state' }, payload => {
        updateScreen(payload.new);
      })
      .subscribe();

    // Realtime 監視 2: comments (弾幕生成)
    supabase
      .channel('admin:comments')
      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'comments' }, payload => {
        if (payload.new && payload.new.content) {
          spawnDanmaku(payload.new.content);
        }
      })
      .subscribe();

    init();
  </script>
